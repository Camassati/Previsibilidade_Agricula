# -*- coding: utf-8 -*-
"""Acesso ao banco MySQL do TCC.

Centraliza a conexão para que os notebooks não precisem repetir credencial.
As credenciais ficam em `banco_config.ini`, que NÃO deve ir para o GitHub nem
ser anexado ao TCC — se o repositório for público, a senha vaza junto.

Uso:
    from banco import conectar, ler_tabela, gravar_tabela

    df = ler_tabela("df_tcc_modelo")
    gravar_tabela(df, "df_tcc_modelo")

Para inspecionar o que existe no banco:
    python banco.py
"""
import configparser
import os
import sys

import pandas as pd

CONFIG = "banco_config.ini"


def _credenciais():
    """Lê de banco_config.ini; variáveis de ambiente têm precedência."""
    cfg = {"host": os.environ.get("TCC_DB_HOST"),
           "user": os.environ.get("TCC_DB_USER"),
           "password": os.environ.get("TCC_DB_PASS"),
           "database": os.environ.get("TCC_DB_NAME"),
           "port": int(os.environ.get("TCC_DB_PORT", 3306))}
    if all(cfg[k] for k in ("host", "user", "password", "database")):
        return cfg

    if not os.path.exists(CONFIG):
        raise FileNotFoundError(
            f"{CONFIG} não encontrado. Crie o arquivo com:\n\n"
            "[mysql]\nhost = ...\nuser = ...\npassword = ...\n"
            "database = ...\nport = 3306\n")
    p = configparser.ConfigParser()
    p.read(CONFIG, encoding="utf-8")
    s = p["mysql"]
    return {"host": s["host"], "user": s["user"], "password": s["password"],
            "database": s["database"], "port": s.getint("port", fallback=3306)}


def conectar():
    """Conexão pymysql. Lembre de fechar, ou use `with conectar() as c:`."""
    import pymysql
    c = _credenciais()
    return pymysql.connect(host=c["host"], user=c["user"], password=c["password"],
                           database=c["database"], port=c["port"],
                           charset="utf8mb4", connect_timeout=30,
                           cursorclass=pymysql.cursors.Cursor)


def engine():
    """Engine SQLAlchemy, que é o que o pandas prefere para to_sql."""
    from sqlalchemy import create_engine
    from urllib.parse import quote_plus
    c = _credenciais()
    return create_engine(
        f"mysql+pymysql://{c['user']}:{quote_plus(c['password'])}"
        f"@{c['host']}:{c['port']}/{c['database']}?charset=utf8mb4",
        pool_pre_ping=True)


def listar_tabelas():
    with conectar() as c, c.cursor() as k:
        k.execute("SHOW TABLES")
        return [r[0] for r in k.fetchall()]


def ler_tabela(nome, limite=None):
    sql = f"SELECT * FROM `{nome}`" + (f" LIMIT {int(limite)}" if limite else "")
    return pd.read_sql(sql, engine())


def gravar_tabela(df, nome, se_existir="replace", lote=1000):
    """Grava o DataFrame numa tabela. `se_existir`: replace | append | fail."""
    df.to_sql(nome, engine(), if_exists=se_existir, index=False,
              chunksize=lote, method="multi")
    print(f"{nome}: {len(df)} linhas gravadas")


# ----------------------------------------------------------- inspeção -------
def inventario(saida="banco_inventario.txt", amostra=3):
    """Varre o banco e escreve um inventário legível: tabelas, colunas, tipos,
    contagem de linhas e uma amostra. É o que eu preciso ver para adaptar o
    pipeline ao que de fato está lá."""
    linhas = []

    def p(s=""):
        print(s)
        linhas.append(str(s))

    with conectar() as c, c.cursor() as k:
        k.execute("SELECT VERSION(), DATABASE()")
        versao, banco = k.fetchone()
        p(f"MySQL {versao} | banco: {banco}")
        k.execute("SHOW TABLES")
        tabelas = [r[0] for r in k.fetchall()]
        p(f"{len(tabelas)} tabelas\n")

        for t in tabelas:
            k.execute(f"SELECT COUNT(*) FROM `{t}`")
            n = k.fetchone()[0]
            p("=" * 72)
            p(f"TABELA {t}  —  {n:,} linhas")
            p("=" * 72)

            k.execute(f"DESCRIBE `{t}`")
            cols = k.fetchall()
            p(f"{len(cols)} colunas:")
            for nome, tipo, nulo, chave, padrao, extra in cols:
                marca = f" [{chave}]" if chave else ""
                p(f"   {nome:34s} {tipo:22s} {'NULL' if nulo == 'YES' else 'NOT NULL':9s}{marca}")

            # chaves estrangeiras, que é o que diz se há integridade referencial
            k.execute("""
                SELECT COLUMN_NAME, REFERENCED_TABLE_NAME, REFERENCED_COLUMN_NAME
                FROM information_schema.KEY_COLUMN_USAGE
                WHERE TABLE_SCHEMA = %s AND TABLE_NAME = %s
                  AND REFERENCED_TABLE_NAME IS NOT NULL""", (banco, t))
            fks = k.fetchall()
            if fks:
                p("chaves estrangeiras:")
                for col, rt, rc in fks:
                    p(f"   {col} -> {rt}.{rc}")
            else:
                p("chaves estrangeiras: nenhuma")

            if n and amostra:
                k.execute(f"SELECT * FROM `{t}` LIMIT {int(amostra)}")
                nomes = [d[0] for d in k.description]
                p(f"amostra ({amostra} linhas):")
                for r in k.fetchall():
                    par = ", ".join(f"{a}={b!r}" for a, b in list(zip(nomes, r))[:8])
                    p(f"   {par}" + (" ..." if len(nomes) > 8 else ""))
            p()

    with open(saida, "w", encoding="utf-8") as f:
        f.write("\n".join(linhas))
    print(f"\n>>> Inventário gravado em {saida}")
    return tabelas


if __name__ == "__main__":
    try:
        inventario()
    except Exception as e:
        print(f"\nFALHOU: {type(e).__name__}: {e}\n", file=sys.stderr)
        print("Causas comuns:", file=sys.stderr)
        print("  - banco_config.ini ausente ou com credencial errada", file=sys.stderr)
        print("  - o DBaaS exige liberar o seu IP no painel antes de aceitar conexão",
              file=sys.stderr)
        print("  - falta instalar: pip install pymysql sqlalchemy pandas", file=sys.stderr)
        sys.exit(1)

# -*- coding: utf-8 -*-
"""Sobe os dados corrigidos do pipeline para o MySQL, sem tocar no IoT.

PROTEGIDAS — o script não escreve nelas e confere no fim se continuam iguais:
    SENSOR_IOT           o ESP32 cadastrado
    LEITURA_IOT          as leituras reais do sensor
    CONFIGURACAO_SAFRA   LEITURA_IOT tem chave estrangeira para cá; mexer quebra

O QUE FAZ — deixa o banco idêntico às bases locais:
    REGIAO                      insere estações que faltam (preserva os
                                id_regiao existentes, porque há FK apontando
                                para eles) e acrescenta id_solo -> TIPO_SOLO
    PONTE_ESTACAO_MUNICIPIO     cria e popula; é a tabela que faltava para ligar
                                estação a município e fechar a integridade
    CLIMA_SEMANAL               cria e popula com df_tcc_semana.xlsx (47 mil)
    DADOS_CLIMATICOS_HISTORICOS backup e recarga com o mensal derivado da base
                                semanal — passa a cobrir 2020-2025 em vez de
                                só 2023 em diante
    PRODUTIVIDADE_OBSERVADA     backup e recarga com a PAM corrigida, safra como
                                ano de verdade em vez do código 112 do SIDRA
    BASE_MODELO                 cria e popula com df_tcc_modelo.xlsx

USO:
    python migrar_banco.py              simula e mostra o que faria, sem escrever
    python migrar_banco.py --aplicar    executa de verdade
"""
import sys
from datetime import datetime

import numpy as np
import pandas as pd

from banco import conectar, engine

ARQ_SEMANA = "df_tcc_semana.xlsx"
ARQ_MODELO = "df_tcc_modelo.xlsx"
PROTEGIDAS = ["SENSOR_IOT", "LEITURA_IOT", "CONFIGURACAO_SAFRA"]
APLICAR = "--aplicar" in sys.argv
LOTE = 1000


def log(s=""):
    print(s, flush=True)


def contar(cur, tabela):
    cur.execute(f"SELECT COUNT(*) FROM `{tabela}`")
    return cur.fetchone()[0]


def existe_coluna(cur, tabela, coluna):
    cur.execute("""SELECT COUNT(*) FROM information_schema.COLUMNS
                   WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = %s
                     AND COLUMN_NAME = %s""", (tabela, coluna))
    return cur.fetchone()[0] > 0


def executar(cur, sql, args=None, descricao=None):
    if descricao:
        log(f"   {descricao}")
    if APLICAR:
        cur.execute(sql, args or ())
    return APLICAR


# --------------------------------------------------------------- leitura ----
log("Lendo as planilhas do pipeline...")
sem = pd.read_excel(ARQ_SEMANA, sheet_name="dados")
xm = pd.ExcelFile(ARQ_MODELO)
modelo = xm.parse("modelo")
pam = xm.parse("pam_bruto")
ponte = xm.parse("ponte_municipio")
log(f"   clima semanal : {len(sem):,} linhas, {sem.Cod_Estacao.nunique()} estações")
log(f"   PAM corrigida : {len(pam):,} linhas, safras {sorted(pam.ano_pam.dropna().unique())}")
log(f"   base do modelo: {len(modelo):,} linhas")

# --------------------------------- semanas partidas na virada do ano ---------
# O ETL agrupou por (ano civil, semana ISO). Uma semana ISO que atravessa 31/12
# aparece então duas vezes com o MESMO periodo: um trecho em dezembro e outro em
# janeiro (ex. A740 / 202152 = 27-31/12/2021 + 01-02/01/2022). Em CLIMA_SEMANAL
# existe UNIQUE KEY (id_regiao, periodo), então os dois trechos têm de ser
# reunidos na semana única que de fato são — com a mesma semântica de agregação
# usada no resto do pipeline.
SEM_SOMA = ["dias_validos", "graus_dia_acum",
            "precipitacao_total_mm", "dias_com_chuva", "radiacao_total_mjm2"]
# dias_no_periodo é propriedade da semana (7), não do trecho: os dois trechos já
# trazem 7 cada um, então somar daria 14. Fica o máximo.
SEM_MAX = ["dias_no_periodo", "temp_max_abs", "maior_veranico",
           "dias_secos_acumulados", "vpd_max_kpa", "vento_rajada_max"]
SEM_MIN = ["temp_min_abs", "umidade_min_abs"]

_chave = ["Cod_Estacao", "periodo"]
_dups = sem[sem.duplicated(_chave, keep=False)]
if len(_dups):
    pares = _dups.groupby(_chave, sort=True).size()
    log(f"   {len(_dups)} linhas em {len(pares)} semanas partidas na virada do "
        f"ano — reunindo em uma linha por semana:")
    for (est, per), n in pares.items():
        log(f"      {est} / {per} ({n} trechos)")

    _num = [c for c in sem.columns
            if pd.api.types.is_numeric_dtype(sem[c]) and c not in ("periodo",)]
    _agg = {}
    for c in sem.columns:
        if c in _chave:
            continue
        if c in SEM_SOMA:
            _agg[c] = lambda x: x.sum(min_count=1)       # acumulados somam
        elif c in SEM_MAX:
            _agg[c] = "max"                              # extremos superiores
        elif c in SEM_MIN:
            _agg[c] = "min"                              # extremos inferiores
        elif c == "data_inicio":
            _agg[c] = "min"
        elif c in ("data_fim", "periodo_label"):
            _agg[c] = "max"
        elif c in _num:
            _agg[c] = "mean"                             # estados médios
        else:
            _agg[c] = "first"                            # UF, Cidade, solo...
    antes = len(sem)
    sem = (sem.groupby(_chave, as_index=False, sort=False).agg(_agg)
              .reindex(columns=sem.columns))
    # 'cobertura' é razão, não estado: recalcula em vez de herdar a média
    if {"dias_validos", "dias_no_periodo"} <= set(sem.columns):
        sem["cobertura"] = (sem.dias_validos /
                            sem.dias_no_periodo.replace(0, np.nan))
    log(f"   clima semanal : {antes:,} -> {len(sem):,} linhas")
    assert not sem.duplicated(_chave).any(), "ainda há (estação, periodo) repetido"
else:
    log("   nenhuma semana partida na virada do ano")

# Guarda: a PAM tem que ter ano de verdade, não o código 112 do SIDRA.
anos = set(pd.to_numeric(pam["ano_pam"], errors="coerce").dropna().astype(int))
if not anos or min(anos) < 1990 or max(anos) > 2100:
    raise SystemExit(
        f"ABORTADO: ano_pam fora de faixa ({sorted(anos)[:5]}). Se aparecer 112, "
        "a planilha é a versão anterior à correção do SIDRA — reexecute o "
        "notebook do alvo antes de migrar.")
log(f"   OK: anos da PAM plausíveis ({min(anos)}–{max(anos)})")

con = conectar()
cur = con.cursor()

# ------------------------------------------------------ estado protegido ----
log("\nConferindo as tabelas protegidas (antes):")
antes = {}
for t in PROTEGIDAS:
    antes[t] = contar(cur, t)
    log(f"   {t:22s} {antes[t]:>7,} linhas")

if not APLICAR:
    log("\n*** SIMULAÇÃO — nada será gravado. Rode com --aplicar para valer. ***")

# ------------------------------------------------------------- 1. REGIAO ----
log("\n1. REGIAO")
cur.execute("SELECT id_regiao, cod_estacao FROM REGIAO")
reg = {c: i for i, c in cur.fetchall() if c}
log(f"   {len(reg)} estações já cadastradas")

est = (sem[["Cod_Estacao", "UF", "Cidade", "id_solo"]]
       .drop_duplicates("Cod_Estacao"))
faltam = est[~est.Cod_Estacao.isin(reg)]
log(f"   {len(faltam)} estações da base semanal ainda não estão no banco")
if len(faltam) and APLICAR:
    cur.executemany(
        "INSERT INTO REGIAO (uf, cidade, cod_estacao) VALUES (%s, %s, %s)",
        [(r.UF, r.Cidade, r.Cod_Estacao) for r in faltam.itertuples()])
    con.commit()
    cur.execute("SELECT id_regiao, cod_estacao FROM REGIAO")
    reg = {c: i for i, c in cur.fetchall() if c}
    log(f"   inseridas; agora {len(reg)} estações")

# liga REGIAO -> TIPO_SOLO, que é o que faz o solo finalmente chegar ao modelo
if not existe_coluna(cur, "REGIAO", "id_solo"):
    executar(cur, "ALTER TABLE REGIAO ADD COLUMN id_solo INT UNSIGNED NULL",
             descricao="acrescenta a coluna id_solo")
    executar(cur, "ALTER TABLE REGIAO ADD CONSTRAINT fk_regiao_solo "
                  "FOREIGN KEY (id_solo) REFERENCES TIPO_SOLO(id_solo)",
             descricao="acrescenta a chave estrangeira para TIPO_SOLO")
else:
    log("   coluna id_solo já existe")

solos = est.dropna(subset=["id_solo"])
log(f"   preenche id_solo em {len(solos)} estações "
    f"({solos.id_solo.nunique()} tipos distintos)")
if APLICAR:
    cur.executemany("UPDATE REGIAO SET id_solo = %s WHERE cod_estacao = %s",
                    [(int(r.id_solo), r.Cod_Estacao) for r in solos.itertuples()])
    con.commit()

# ------------------------------------------------------ 2. CLIMA_SEMANAL ----
log("\n2. CLIMA_SEMANAL")
NUM = [c for c in sem.columns if c not in
       ("periodo", "periodo_label", "data_inicio", "data_fim", "UF", "Cidade",
        "Cod_Estacao", "id_solo", "nome_solo", "ordem_solo", "textura", "fonte",
        "culturas_predominantes", "retencao_agua", "drenagem", "aptidao_graos",
        "profundidade", "risco_erosao")
       and pd.api.types.is_numeric_dtype(sem[c])]
log(f"   {len(NUM)} colunas climáticas")

clima = sem[["Cod_Estacao", "periodo", "periodo_label", "data_inicio", "data_fim"] + NUM].copy()
clima["id_regiao"] = clima.Cod_Estacao.map(reg)
orfas = int(clima.id_regiao.isna().sum())
if orfas:
    log(f"   {orfas} linhas sem id_regiao — serão descartadas")
    clima = clima.dropna(subset=["id_regiao"])
clima["id_regiao"] = clima.id_regiao.astype(int)
clima = clima.drop(columns="Cod_Estacao")
clima = clima.replace([np.inf, -np.inf], np.nan).where(pd.notna(clima), None)
log(f"   {len(clima):,} linhas prontas para gravar")

cols_sql = ",\n        ".join(f"`{c}` DECIMAL(12,3) NULL" for c in NUM)
DDL_CLIMA = f"""
    CREATE TABLE CLIMA_SEMANAL (
        id_clima_sem BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
        id_regiao    INT UNSIGNED NOT NULL,
        periodo      INT UNSIGNED NOT NULL COMMENT 'ano ISO * 100 + semana ISO',
        periodo_label VARCHAR(10) NOT NULL COMMENT 'ex. 2023-W07',
        data_inicio  DATE NULL,
        data_fim     DATE NULL,
        {cols_sql},
        UNIQUE KEY uq_regiao_periodo (id_regiao, periodo),
        KEY ix_periodo (periodo),
        CONSTRAINT fk_clima_sem_regiao FOREIGN KEY (id_regiao)
            REFERENCES REGIAO(id_regiao)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4"""
executar(cur, "DROP TABLE IF EXISTS CLIMA_SEMANAL", descricao="dropa se já existir")
executar(cur, DDL_CLIMA, descricao=f"cria a tabela ({len(NUM) + 6} colunas)")

# ----------------------------------------- 3. PRODUTIVIDADE_OBSERVADA -------
log("\n3. PRODUTIVIDADE_OBSERVADA")
bkp = f"PRODUTIVIDADE_OBSERVADA_bkp_{datetime.now():%Y%m%d}"
n_atual = contar(cur, "PRODUTIVIDADE_OBSERVADA")
log(f"   {n_atual:,} linhas atuais (com safra='112') -> backup em {bkp}")
executar(cur, f"DROP TABLE IF EXISTS `{bkp}`")
executar(cur, f"RENAME TABLE PRODUTIVIDADE_OBSERVADA TO `{bkp}`",
         descricao="renomeia a atual para backup")
executar(cur, """
    CREATE TABLE PRODUTIVIDADE_OBSERVADA (
        id_produtividade  BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
        cod_municipio_ibge INT UNSIGNED NOT NULL,
        municipio_ibge    VARCHAR(150) NULL,
        uf                CHAR(2) NOT NULL,
        cultura_tcc       VARCHAR(100) NOT NULL,
        safra             SMALLINT UNSIGNED NOT NULL COMMENT 'ano da safra PAM',
        rendimento_kg_ha   DECIMAL(12,2) NULL,
        rendimento_sacas_ha DECIMAL(10,2) NULL,
        UNIQUE KEY uq_mun_cultura_safra (cod_municipio_ibge, cultura_tcc, safra),
        KEY ix_uf_safra (uf, safra)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4""",
         descricao="recria com safra SMALLINT e chave única")

SACAS = {"Soja (em grão)": 60, "Milho (em grão)": 60, "Trigo (em grão)": 60}
# pam_bruto já traz 'cultura' E 'cultura_tcc' (o nome bruto do SIDRA e o
# normalizado). Renomear 'cultura' criaria duas colunas com o mesmo nome, e aí
# df['cultura_tcc'] devolve um DataFrame em vez de uma Series.
p = pam.drop(columns=[c for c in ("cultura",) if "cultura_tcc" in pam.columns
                      and c in pam.columns]).copy()
renomear = {"cod_municipio": "cod_municipio_ibge", "ano_pam": "safra"}
if "cultura_tcc" not in p.columns:
    renomear["cultura"] = "cultura_tcc"
p = p.rename(columns=renomear)
assert not p.columns.duplicated().any(), \
    f"colunas duplicadas após o rename: {list(p.columns[p.columns.duplicated()])}"
p["uf"] = p["UF"] if "UF" in p.columns else None
p["safra"] = pd.to_numeric(p.safra, errors="coerce").astype("Int64")
p["cod_municipio_ibge"] = pd.to_numeric(p.cod_municipio_ibge, errors="coerce").astype("Int64")
p["rendimento_sacas_ha"] = p.rendimento_kg_ha / p.cultura_tcc.map(SACAS).fillna(60)
p = p.dropna(subset=["cod_municipio_ibge", "safra", "cultura_tcc"])
antes_dedup = len(p)
p = p.drop_duplicates(["cod_municipio_ibge", "cultura_tcc", "safra"])
log(f"   {antes_dedup:,} linhas -> {len(p):,} após remover duplicatas "
    f"({antes_dedup - len(p):,} repetidas)")
prod = p[["cod_municipio_ibge", "municipio_ibge", "uf", "cultura_tcc", "safra",
          "rendimento_kg_ha", "rendimento_sacas_ha"]].where(pd.notna(p), None)

# ------------------------------- 3b. DADOS_CLIMATICOS_HISTORICOS (mensal) ----
log("\n3b. DADOS_CLIMATICOS_HISTORICOS")
# O mensal é derivado do semanal, com a mesma semântica de agregação: soma para
# acumulados, média para estados, extremos para máx/mín. Cada semana é atribuída
# ao mês da sua data de início — aproximação necessária, porque uma semana ISO
# pode atravessar a virada do mês.
MENSAL_SOMA = ["precipitacao_total_mm", "graus_dia_acum", "radiacao_total_mjm2",
               "dias_com_chuva", "dias_validos"]
MENSAL_MEDIA = ["temp_media", "temp_max_media", "temp_min_media", "amplitude_termica",
                "umidade_media", "vpd_medio_kpa", "orvalho_medio", "pressao_media",
                "vento_medio", "vento_dir_predom", "vento_dir_consistencia"]
MENSAL_MAX = ["temp_max_abs", "maior_veranico", "vento_rajada_max"]
MENSAL_MIN = ["temp_min_abs", "umidade_min_abs"]

mes = sem.copy()
mes["ano_mes"] = pd.to_datetime(mes["data_inicio"]).dt.to_period("M").dt.to_timestamp()
agg = {c: (lambda x: x.sum(min_count=1)) for c in MENSAL_SOMA if c in mes.columns}
agg.update({c: "mean" for c in MENSAL_MEDIA if c in mes.columns})
agg.update({c: "max" for c in MENSAL_MAX if c in mes.columns})
agg.update({c: "min" for c in MENSAL_MIN if c in mes.columns})
mensal = mes.groupby(["Cod_Estacao", "ano_mes"], as_index=False).agg(agg)
mensal["dias_no_mes"] = mensal.ano_mes.dt.days_in_month
mensal["cobertura"] = (mensal.dias_validos / mensal.dias_no_mes).round(2).clip(upper=1)
mensal["id_regiao"] = mensal.Cod_Estacao.map(reg)
mensal = mensal.dropna(subset=["id_regiao"])
mensal["id_regiao"] = mensal.id_regiao.astype(int)
mensal = mensal.drop(columns="Cod_Estacao")
mensal = mensal.replace([np.inf, -np.inf], np.nan).where(pd.notna(mensal), None)
log(f"   {len(mensal):,} linhas mensais, "
    f"{mes.ano_mes.min():%Y-%m} a {mes.ano_mes.max():%Y-%m}")

bkp_clima = f"DADOS_CLIMATICOS_HISTORICOS_bkp_{datetime.now():%Y%m%d}"

# -*- coding: utf-8 -*-
"""Calculadora Agrícola — núcleo de cálculo.

Vai do rendimento municipal previsto pelo Elastic Net até o resultado do
produtor: sacas na área dele, receita, custo, lucro e risco de ficar abaixo
do ponto de equilíbrio.

Usado pelo notebook (ipywidgets) e pela página HTML. Toda a conta está aqui,
para que as duas interfaces nunca divirjam.
"""
import json
import numpy as np

# Referências editáveis pelo usuário. Valores de outubro/2026, para ancorar
# o formulário — o produtor informa os dele.
PRECO_REF = {            # R$ por saca de 60 kg
    "Soja (em grão)":  154.35,   # CEPEA/ESALQ, Paraná interior, 01/10/2026
    "Milho (em grão)":  62.00,   # ordem de grandeza do indicador CEPEA
    "Trigo (em grão)":  92.30,   # CEPEA/ESALQ PR, R$ 1.538,31/t convertido
}
CUSTO_REF = {            # R$ por hectare, custo de produção
    "Soja (em grão)":  6115.83,  # Aprosoja/MS, safra 2025/26
    "Milho (em grão)": 5200.00,
    "Trigo (em grão)": 4100.00,
}


class Calculadora:
    def __init__(self, caminho="modelo_export.json"):
        with open(caminho, encoding="utf-8") as f:
            self.d = json.load(f)
        self.ordem = self.d["ordem_m3"]
        self.pos = {v: i for i, v in enumerate(self.ordem)}
        self.linhas = self.d["linhas"]
        self.valores = self.d["valores"]

    # ---------------------------------------------------------------- modelo
    def prever_rendimento(self, idx, momento="M3"):
        """Rendimento municipal previsto, em sacas/ha, com a contribuição de
        cada variável. É um Elastic Net: previsão é soma de termos lineares,
        e é por isso que dá para abrir o resultado variável a variável."""
        m = self.d["momentos"][momento]
        linha = self.valores[idx]
        total = m["intercept"]
        contrib = []
        for j, var in enumerate(m["variaveis"]):
            coef = m["coef"][j]
            bruto = linha[self.pos[var]]
            if bruto is None:
                bruto = m["mediana"][j]
            # mesma escala Min-Max que o Pipeline aplicou no treino
            esc = (bruto - m["min"][j]) / m["range"][j]
            esc = min(1.0, max(0.0, esc))
            parcela = coef * esc
            total += parcela
            if coef != 0:
                contrib.append({"variavel": var, "valor": bruto,
                                "coeficiente": coef, "contribuicao": parcela})
        contrib.sort(key=lambda c: -abs(c["contribuicao"]))
        return total, contrib

    # ----------------------------------------------------------- calibração
    def rendimento_municipal_real(self, cod_municipio, cultura, ano):
        """Rendimento que o município de fato teve naquela safra (PAM/IBGE)."""
        for l in self.linhas:
            if l["m"] == cod_municipio and l["c"] == cultura and l["s"] == ano:
                return l["real"]
        return None

    def fator_produtor(self, historico, cod_municipio, cultura):
        """Quanto o produtor rende acima ou abaixo do município dele.

        `historico` é uma lista de pares (ano, rendimento_dele). Cada safra
        informada é comparada ao rendimento REAL do município NAQUELE MESMO
        ano — comparar com outro ano mediria a diferença entre safras, não a
        diferença entre o produtor e a região dele.

        Com menos de duas safras não há fator: uma só é anedota, não padrão.
        """
        razoes, usadas, ignoradas = [], [], []
        for ano, rend in historico:
            mun = self.rendimento_municipal_real(cod_municipio, cultura, ano)
            if mun and mun > 0:
                razoes.append(rend / mun)
                usadas.append({"ano": ano, "produtor": rend, "municipio": mun,
                               "razao": rend / mun})
            else:
                ignoradas.append(ano)
        if len(razoes) < 2:
            return 1.0, usadas, ignoradas, None
        return float(np.mean(razoes)), usadas, ignoradas, float(np.std(razoes))

    # Faixa plausível para o fator. Fora dela, quase sempre é erro de unidade:
    # o produtor informou sacas da área toda em vez de sacas por hectare, ou
    # quilos em vez de sacas. Vale avisar em vez de aplicar um fator absurdo.
    FATOR_MIN, FATOR_MAX = 0.55, 1.80

    @classmethod
    def fator_suspeito(cls, fator, usadas):
        if fator == 1.0 or not usadas:
            return None
        if fator > cls.FATOR_MAX:
            return (f"O histórico informado dá um fator de {fator:.2f}x — o produtor "
                    f"renderia {fator:.1f} vezes a média do município dele. É possível, "
                    "mas o mais comum é o valor ter sido digitado em sacas da área "
                    "toda, e não em sacas por hectare. Confira antes de usar.")
        if fator < cls.FATOR_MIN:
            return (f"O histórico informado dá um fator de {fator:.2f}x — o produtor "
                    "renderia bem abaixo da média do município. Confira se os valores "
                    "estão em sacas por hectare.")
        return None

    # ------------------------------------------------------------- resultado
    def calcular(self, idx, area_ha, momento="M3", historico=None,
                 preco=None, custo_ha=None):
        linha = self.linhas[idx]
        cult = linha["c"]
        rend_mun, contrib = self.prever_rendimento(idx, momento)
        fator, usadas, ignoradas, disp = self.fator_produtor(
            historico or [], linha["m"], cult)
        rend = rend_mun * fator

        preco = PRECO_REF[cult] if preco is None else preco
        custo_ha = CUSTO_REF[cult] if custo_ha is None else custo_ha

        res = self.d["momentos"][momento]["residuo"][cult]
        todos = np.array(res["todos"]) * fator   # resíduo escala com o fator

        # resíduo = previsto - realizado, então realizado = previsto - resíduo
        faixa = {
            "p10": rend - np.percentile(todos, 90),
            "p25": rend - np.percentile(todos, 75),
            "p50": rend - np.percentile(todos, 50),
            "p75": rend - np.percentile(todos, 25),
            "p90": rend - np.percentile(todos, 10),
        }
        equilibrio = custo_ha / preco if preco else float("inf")
        # distribuição empírica do realizado, para a probabilidade de prejuízo
        realizados = rend - todos
        p_abaixo = float((realizados < equilibrio).mean())

        def financeiro(r_ha):
            sacas = r_ha * area_ha
            receita = sacas * preco
            custo = custo_ha * area_ha
            return {"rendimento_sacas_ha": r_ha, "sacas_totais": sacas,
                    "receita": receita, "custo": custo, "lucro": receita - custo,
                    "margem_pct": 100 * (receita - custo) / receita if receita else 0.0}

        return {
            "municipio": linha["nome"], "uf": linha["uf"], "cultura": cult,
            "safra": linha["s"], "area_ha": area_ha, "momento": momento,
            "rendimento_municipal": rend_mun,
            "fator_produtor": fator, "safras_usadas": usadas,
            "safras_ignoradas": ignoradas, "dispersao_fator": disp,
            "preco_saca": preco, "custo_hectare": custo_ha,
            "ponto_equilibrio_sacas_ha": equilibrio,
            "prob_abaixo_equilibrio": p_abaixo,
            "central": financeiro(rend),
            "pessimista": financeiro(faixa["p10"]),
            "provavel_baixo": financeiro(faixa["p25"]),
            "provavel_alto": financeiro(faixa["p75"]),
            "otimista": financeiro(faixa["p90"]),
            "realizado_sacas_ha": linha["real"],
            "media_historica_sacas_ha": linha["hist"],
            "contribuicoes": contrib[:12],
        }


def brl(v):
    s = f"{abs(v):,.2f}".replace(",", "~").replace(".", ",").replace("~", ".")
    return ("-R$ " if v < 0 else "R$ ") + s

                df["rendimento_sacas_ha"].between(lo, hi))
df = df[mascara].copy()
print(f"Após filtros: {len(df)} linhas | {df['cod_municipio'].nunique()} municípios")

# Guarda de sanidade: a mesma do notebook de modelagem. Zeros impossíveis aqui
# significam que o notebook do alvo rodou sem agregar_coluna(min_count=1).
impossivel = [c for c in df.columns
              if c.endswith(("graus_dia_acum", "radiacao_total_mjm2"))
              and not c.startswith("ciclo_")]
com_zero = int((df[impossivel] == 0).any(axis=1).sum())
if com_zero:
    raise AssertionError(
        f"{com_zero} linhas com zero impossível em {len(impossivel)} colunas. "

g = df.groupby(["cod_municipio", "cultura_tcc"])["rendimento_sacas_ha"]
df["base_hist"] = g.apply(lambda s: s.shift().expanding().mean()).reset_index(level=[0,1], drop=True)
ufp = (df.sort_values("ano_pam").groupby(["UF","cultura_tcc"])["rendimento_sacas_ha"]
         .apply(lambda s: s.shift().expanding().mean()).reset_index(level=[0,1], drop=True))

SOLO = ["retencao_agua","drenagem","aptidao_graos","profundidade","risco_erosao"]
MOM = {"M1":["vegetativo_"], "M2":["vegetativo_","floracao_"],
       "M3":["vegetativo_","floracao_","enchimento_"]}
def cols(pref):
    return [c for c in df.columns if any(c.startswith(p) for p in pref)] + \
           [c for c in SOLO if c in df.columns] + ["base_hist"]

y = df.rendimento_sacas_ha.values.astype(float)
gr = df.cod_municipio.values
GRADE = {"m__alpha":[0.001,0.01,0.1,0.5,1.0,5.0],
         "m__l1_ratio":[0.1,0.3,0.5,0.7,0.9,0.95]}

    EXPORT["momentos"][nome] = {
        "variaveis": cs,
        "min": [float(v) for v in esc.data_min_],
        "range": [float(v) if v else 1.0 for v in (esc.data_max_-esc.data_min_)],
        "coef": [round(float(v),6) for v in mod.coef_],
        "intercept": float(mod.intercept_),
        "mediana": [float(med.get(c,0.0)) for c in cs],
        "alpha": float(fin.best_params_["m__alpha"]),
        "l1_ratio": float(fin.best_params_["m__l1_ratio"]),
        "rmse_oos": round(rmse,3),
        "zeradas": int((mod.coef_==0).sum()),
    }
    # quantis do resíduo relativo, por cultura

                     "todos":[round(float(v),2) for v in np.sort(r)]}
    EXPORT["momentos"][nome]["residuo"] = res
    print(f"{nome}: {len(cs)} vars | RMSE oos {rmse:.2f} | alpha {fin.best_params_['m__alpha']} "

"""Análise de robustez — os três testes que qualificam a conclusão.

Roda depois da modelagem e responde três perguntas que a banca pode fazer e
que a tabela de desempenho sozinha não responde:

  1. O ganho sobre o método tradicional sobrevive a um baseline que recebe a

  2. O modelo acerta a safra de quebra quando precisa PREVÊ-LA, e não apenas

  3. O valor do modelo se concentra nos anos de clima atípico?

                "Trigo (em grão)": (10, 100)}

# O caso que a banca vai perguntar: a seca do Rio Grande do Sul.

# ------------------------------------------------------------------ carga ---
def carregar():
    df = pd.read_excel(ARQUIVO, sheet_name="modelo")
    df = df[df["cobertura_ciclo"] >= MIN_COBERTURA_CICLO].copy()
    m = pd.Series(False, index=df.index)
    for cultura, (lo, hi) in FAIXAS_SACAS.items():
        m |= (df["cultura_tcc"] == cultura) & df["rendimento_sacas_ha"].between(lo, hi)
    df = df[m].sort_values(["cod_municipio", "cultura_tcc", "ano_pam"]).reset_index(drop=True)

    g = df.groupby(["cod_municipio", "cultura_tcc"])["rendimento_sacas_ha"]
    df["base_hist"] = (g.apply(lambda s: s.shift().expanding().mean())
                        .reset_index(level=[0, 1], drop=True))
    uf = (df.sort_values("ano_pam").groupby(["UF", "cultura_tcc"])["rendimento_sacas_ha"]
            .apply(lambda s: s.shift().expanding().mean())
            .reset_index(level=[0, 1], drop=True))
    df["base_hist"] = df.base_hist.fillna(uf).fillna(df.rendimento_sacas_ha.mean())

    # Método tradicional BEM ESPECIFICADO: a média do município escalada pelo
    # fator do ano na região, estimado leave-one-out nos demais municípios.
    k = ["UF", "cultura_tcc", "ano_pam"]
    sr = df.groupby(k)["rendimento_sacas_ha"].transform("sum")
    sh = df.groupby(k)["base_hist"].transform("sum")
    fator = ((sr - df.rendimento_sacas_ha) / (sh - df.base_hist).replace(0, np.nan))
    df["base_indexada"] = (df.base_hist * fator.fillna(1.0)).fillna(df.base_hist)
    return df


def modelos(semente=SEMENTE):
    return {
        "elastic_net": (
            Pipeline([("e", MinMaxScaler()),
                      ("m", ElasticNet(max_iter=20000, random_state=semente))]),
            {"m__alpha": [0.001, 0.01, 0.1, 0.5, 1.0, 5.0],
             "m__l1_ratio": [0.1, 0.3, 0.5, 0.7, 0.9, 0.95]}),
        "random_forest": (
            Pipeline([("m", RandomForestRegressor(n_estimators=200, random_state=semente,
                                                  n_jobs=-1))]),
            {"m__max_depth": [8, 15, None], "m__min_samples_leaf": [1, 5, 10]}),
    }


def prever(df, esquema):
    """Previsões fora da amostra em cada esquema de validação."""
    cols = ([c for c in df.columns
             if c.startswith(("vegetativo_", "floracao_", "enchimento_"))]
            + [c for c in COLUNAS_SOLO if c in df.columns] + ["base_hist"])
    X, y = df[cols], df.rendimento_sacas_ha.values.astype(float)
    gr, idx = df.cod_municipio.values, np.arange(len(df))
    safras = np.sort(df.ano_pam.unique())

    def particoes():
        if esquema == "espacial":
            yield from GroupKFold(n_splits=5).split(idx, groups=gr)
        else:
            for s in safras[1:]:
                tr, te = idx[df.ano_pam.values < s], idx[df.ano_pam.values == s]
                if len(tr) >= 50 and len(te):
                    yield tr, te

    saida = {}
    for nome, (pipe, grade) in modelos().items():
        pv = np.full(len(y), np.nan)
        for tr, te in particoes():
            med = X.iloc[tr].median(numeric_only=True)
            b = GridSearchCV(pipe, grade, cv=GroupKFold(n_splits=4),
                             scoring="neg_root_mean_squared_error", n_jobs=-1)
            b.fit(X.iloc[tr].fillna(med), y[tr], groups=gr[tr])
            pv[te] = b.predict(X.iloc[te].fillna(med))
        saida[nome] = pv
        print(f"   {esquema}/{nome}: pronto")
    return saida


def rmse(p, y):
    p, y = np.asarray(p, float), np.asarray(y, float)
    ok = ~np.isnan(p)
    return float(np.sqrt(((p[ok] - y[ok]) ** 2).mean()))

def rotular(nome):
    """Traduz o nome técnico da coluna para leitura humana, sem inventar: o
    prefixo é o estádio fenológico e o resto é a variável climática."""
    if nome == "base_hist":
        return "histórico do município"
    for pref, leg in ESTADIO.items():
        if nome.startswith(pref + "_"):
            resto = nome[len(pref) + 1:]
            return f"{leg} · {VARIAVEL.get(resto, resto.replace('_', ' '))}"
    return VARIAVEL.get(nome, nome.replace("_", " "))

def parse(texto):
    """Quebra o inventário em {tabela: {colunas, fks, n_linhas}}."""
    tabelas = {}
    blocos = re.split(r"={10,}\n", texto)
    i = 0
    while i < len(blocos):
        cab = blocos[i].strip()
        m = re.match(r"TABELA (\w+)\s+—\s+([\d,]+) linhas", cab)
        if m and i + 1 < len(blocos):
            nome, n = m.group(1), m.group(2)
            corpo = blocos[i + 1]
            cols = []
            for linha in corpo.splitlines():
                c = re.match(r"\s{3}(\w+)\s{2,}([\w()\s,]+?)\s{2,}"
                             r"(NOT NULL|NULL)\s*(?:\[(\w+)\])?\s*$", linha)
                if c:
                    cols.append({"nome": c.group(1), "tipo": c.group(2).strip(),
                                 "nulo": c.group(3), "chave": c.group(4)})
            fks = dict(re.findall(r"\s{3}(\w+) -> ([\w.]+)", corpo))
            tabelas[nome] = {"colunas": cols, "fks": fks, "linhas": n}
        i += 1
    return tabelas

# -*- coding: utf-8 -*-
"""Exporta as leituras do nó sensor para sensor.json, que a calculadora lê.

A página publicada não alcança o MySQL — o navegador do artefato só faz HTTPS
para domínios liberados, e o banco fala protocolo binário na porta 3306. Então
o caminho é este: você roda isto na sua máquina, que enxerga o banco, e o JSON
gerado acompanha a página.

    python exportar_sensor.py              # grava sensor.json
    python exportar_sensor.py --limite 500 # só as N leituras mais recentes

Para manter atualizado, agende no Agendador de Tarefas do Windows.
"""
import argparse
import json
import sys
from datetime import datetime

import pandas as pd

from banco import conectar

AMOSTRAS_SERIE = 240   # pontos no gráfico; o resto vira resumo


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--limite", type=int, default=5000,
                    help="leituras mais recentes a considerar (padrão 5000)")
    ap.add_argument("--saida", default="sensor.json")
    args = ap.parse_args()

    con = conectar()
    # Traz a leitura junto do contexto: qual sensor, qual safra, qual região.
    sql = """
        SELECT l.id_leitura, l.temperatura, l.umidade_relativa, l.data_hora,
               s.modelo, s.fabricante, s.protocolo_comunicacao,
               s.frequencia_amostragem_min, s.status_operacional,
               r.uf, r.cidade, r.cod_estacao,
               c.data_plantio, cu.nome_cultura, t.nome_solo
          FROM LEITURA_IOT l
          JOIN SENSOR_IOT s          ON s.id_sensor  = l.id_sensor
          JOIN CONFIGURACAO_SAFRA c  ON c.id_config  = l.id_config
          LEFT JOIN REGIAO r         ON r.id_regiao  = c.id_regiao
          LEFT JOIN CULTURA cu       ON cu.id_cultura = c.id_cultura
          LEFT JOIN TIPO_SOLO t      ON t.id_solo    = c.id_solo
         ORDER BY l.data_hora DESC
         LIMIT %s"""
    df = pd.read_sql(sql, con, params=(args.limite,))
    con.close()

    if df.empty:
        sys.exit("ABORTADO: LEITURA_IOT não devolveu nenhuma linha.")

    df = df.sort_values("data_hora")
    df["temperatura"] = pd.to_numeric(df.temperatura, errors="coerce")
    df["umidade_relativa"] = pd.to_numeric(df.umidade_relativa, errors="coerce")

    ult = df.iloc[-1]
    inicio, fim = df.data_hora.iloc[0], df.data_hora.iloc[-1]
    duracao_s = (fim - inicio).total_seconds()

    # Intervalo REAL entre leituras, medido — não o que a coluna
    # frequencia_amostragem_min declara. As duas podem divergir, e a página
    # mostra as duas justamente para que a divergência apareça.
    intervalos = df.data_hora.diff().dt.total_seconds().dropna()
    intervalo_medido = float(intervalos.median()) if len(intervalos) else None

    # Reamostra a série para caber na página sem perder a forma.
    passo = max(1, len(df) // AMOSTRAS_SERIE)
    s = df.iloc[::passo]

    out = {
        "gerado_em": datetime.now().isoformat(timespec="seconds"),
        "sensor": {
            "modelo": ult.modelo,
            "fabricante": ult.fabricante,
            "protocolo": ult.protocolo_comunicacao,
            "frequencia_declarada_min": (None if pd.isna(ult.frequencia_amostragem_min)
                                         else int(ult.frequencia_amostragem_min)),
            "intervalo_medido_s": (None if intervalo_medido is None
                                   else round(intervalo_medido, 1)),
            "status": ult.status_operacional,
        },
        "local": {
            "uf": ult.uf, "cidade": ult.cidade, "cod_estacao": ult.cod_estacao,
            "cultura": ult.nome_cultura, "solo": ult.nome_solo,
            "data_plantio": (None if pd.isna(ult.data_plantio)
                             else str(ult.data_plantio)),
        },
        "janela": {
            "inicio": inicio.isoformat(timespec="seconds"),
            "fim": fim.isoformat(timespec="seconds"),
            "leituras": int(len(df)),
            "duracao_horas": round(duracao_s / 3600, 2),
        },
        "ultima": {
            "temperatura": (None if pd.isna(ult.temperatura) else float(ult.temperatura)),
            "umidade": (None if pd.isna(ult.umidade_relativa) else float(ult.umidade_relativa)),
            "quando": ult.data_hora.isoformat(timespec="seconds"),
        },
        "resumo": {
            "temp_media": round(float(df.temperatura.mean()), 2),
            "temp_min": round(float(df.temperatura.min()), 2),
            "temp_max": round(float(df.temperatura.max()), 2),
            "umid_media": round(float(df.umidade_relativa.mean()), 2),
            "umid_min": round(float(df.umidade_relativa.min()), 2),
            "umid_max": round(float(df.umidade_relativa.max()), 2),
        },
        "serie": [
            {"t": r.data_hora.isoformat(timespec="seconds"),
             "temp": (None if pd.isna(r.temperatura) else round(float(r.temperatura), 2)),
             "umid": (None if pd.isna(r.umidade_relativa) else round(float(r.umidade_relativa), 2))}
            for r in s.itertuples()
        ],
    }

    with open(args.saida, "w", encoding="utf-8") as f:
        json.dump(out, f, ensure_ascii=False, indent=1)

    print(f"{args.saida} gerado.")
    print(f"  leituras            : {out['janela']['leituras']:,}")
    print(f"  janela              : {inicio} a {fim} ({out['janela']['duracao_horas']} h)")
    print(f"  pontos no grafico   : {len(out['serie'])}")
    print(f"  temperatura         : {out['resumo']['temp_min']} a {out['resumo']['temp_max']} C")
    print(f"  umidade             : {out['resumo']['umid_min']} a {out['resumo']['umid_max']} %")
    fd = out["sensor"]["frequencia_declarada_min"]
    im = out["sensor"]["intervalo_medido_s"]
    if fd is not None and im is not None:
        print(f"  intervalo declarado : {fd} (coluna frequencia_amostragem_min)")
        print(f"  intervalo medido    : {im} s")
        if abs(fd * 60 - im) > max(5.0, 0.2 * fd * 60):
            print("  ATENCAO: o intervalo medido nao bate com o declarado na coluna.")


if __name__ == "__main__":
    main()
